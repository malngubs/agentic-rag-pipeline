Basic Integration

<!-- Add to your website's <head> section -->
<script src="https://your-domain.com/widget/macrocomm-bubble.js"></script>

<!-- Initialize the widget -->
<script>
MacrocommBubble({
    host: 'https://your-api-domain.com',
    tenant: 'your-company-id'
});
</script>

Custom Branding

<script>
MacrocommBubble({
    host: 'https://your-api-domain.com',
    tenant: 'your-company-id',
    theme: {
        primary: '#FF6E00',      // Your brand color
        secondary: '#FF923F',    // Accent color
        background: '#FFFFFF'    // Background color
    },
    position: 'bottom-left',     // Widget position
    welcomeMessage: 'Hi! How can I help you today?',
    placeholder: 'Ask me anything...'
});
</script>

Enterprise Configuration

<script>
MacrocommBubble({
    host: 'https://your-api-domain.com',
    tenant: 'enterprise-client',
    theme: {
        primary: '#007bff',
        secondary: '#0056b3'
    },
    features: {
        enableFileUpload: true,
        enableVoice: false,
        enableAnalytics: true
    },
    authentication: {
        required: true,
        provider: 'oauth2'
    },
    limits: {
        maxMessageLength: 4000,
        rateLimitRpm: 60
    }
});
</script>
